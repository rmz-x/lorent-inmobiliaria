-- =====================================================
-- SCRIPT LIMPIO PARA LORENT INMOBILIARIA
-- =====================================================

-- 1. TABLAS BASE (sin dependencias)
CREATE TABLE IF NOT EXISTS public.cache (
    key character varying(255) NOT NULL,
    value text NOT NULL,
    expiration integer NOT NULL,
    CONSTRAINT cache_pkey PRIMARY KEY (key)
);

CREATE TABLE IF NOT EXISTS public.cache_locks (
    key character varying(255) NOT NULL,
    owner character varying(255) NOT NULL,
    expiration integer NOT NULL,
    CONSTRAINT cache_locks_pkey PRIMARY KEY (key)
);

CREATE TABLE IF NOT EXISTS public.categorias (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre character varying(100) NOT NULL,
    descripcion text,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.propietarios (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre character varying(255) NOT NULL,
    telefono character varying(20),
    email character varying(255),
    direccion text,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP
);

-- 2. TABLA USUARIOS
CREATE TABLE IF NOT EXISTS public.usuarios (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre character varying(255) NOT NULL,
    correo character varying(255) NOT NULL UNIQUE,
    usuario character varying(255) NOT NULL UNIQUE,
    contrasena character varying(255) NOT NULL,
    rol character varying(255) NOT NULL DEFAULT 'cliente',
    created_at timestamp(0) without time zone,
    updated_at timestamp(0) without time zone,
    deleted_at timestamp(0) without time zone,
    estado character varying(20) NOT NULL DEFAULT 'activo',
    CONSTRAINT usuarios_estado_check CHECK (estado IN ('activo', 'inactivo')),
    CONSTRAINT usuarios_rol_check CHECK (rol IN ('administrador', 'agente', 'asistente', 'cliente'))
);

-- 3. TABLA PROPIEDADES
CREATE TABLE IF NOT EXISTS public.propiedades (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titulo character varying(255) NOT NULL,
    tipo character varying(255) NOT NULL,
    zona character varying(255) NOT NULL,
    precio numeric(12,2) NOT NULL,
    area numeric(8,2) NOT NULL,
    descripcion text NOT NULL,
    estado character varying(255) NOT NULL DEFAULT 'Disponible',
    agente_id bigint REFERENCES public.usuarios(id) ON DELETE CASCADE,
    created_at timestamp(0) without time zone,
    updated_at timestamp(0) without time zone,
    deleted_at timestamp(0) without time zone,
    imagen character varying(255),
    categoria_id bigint REFERENCES public.categorias(id) ON DELETE SET NULL,
    propietario_id bigint REFERENCES public.propietarios(id) ON DELETE SET NULL,
    habitaciones integer DEFAULT 0,
    banos integer DEFAULT 0,
    antiguedad integer DEFAULT 0,
    latitud numeric(10,7),
    longitud numeric(10,7),
    CONSTRAINT propiedades_estado_check CHECK (estado IN ('Disponible', 'Reservado', 'Vendido')),
    CONSTRAINT propiedades_tipo_check CHECK (tipo IN ('Venta', 'Alquiler', 'Anticretico'))
);

-- 4. TABLAS DE JOBS Y MIGRACIONES
CREATE TABLE IF NOT EXISTS public.jobs (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    queue character varying(255) NOT NULL,
    payload text NOT NULL,
    attempts smallint NOT NULL,
    reserved_at integer,
    available_at integer NOT NULL,
    created_at integer NOT NULL
);

CREATE TABLE IF NOT EXISTS public.failed_jobs (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    uuid character varying(255) NOT NULL UNIQUE,
    connection text NOT NULL,
    queue text NOT NULL,
    payload text NOT NULL,
    exception text NOT NULL,
    failed_at timestamp(0) without time zone NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.job_batches (
    id character varying(255) NOT NULL PRIMARY KEY,
    name character varying(255) NOT NULL,
    total_jobs integer NOT NULL,
    pending_jobs integer NOT NULL,
    failed_jobs integer NOT NULL,
    failed_job_ids text NOT NULL,
    options text,
    cancelled_at integer,
    created_at integer NOT NULL,
    finished_at integer
);

CREATE TABLE IF NOT EXISTS public.migrations (
    id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    migration character varying(255) NOT NULL,
    batch integer NOT NULL
);

-- 5. ÍNDICES (DESPUÉS DE CREAR LAS TABLAS)
CREATE INDEX IF NOT EXISTS cache_expiration_index ON public.cache (expiration);
CREATE INDEX IF NOT EXISTS cache_locks_expiration_index ON public.cache_locks (expiration);
CREATE INDEX IF NOT EXISTS jobs_queue_index ON public.jobs (queue);

-- 6. RESTO DE TABLAS
CREATE TABLE IF NOT EXISTS public.contratos (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    cliente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    agente_id bigint REFERENCES public.usuarios(id) ON DELETE SET NULL,
    tipo_contrato character varying(50) NOT NULL,
    monto numeric(12,2) NOT NULL,
    fecha_inicio date NOT NULL,
    fecha_fin date,
    estado character varying(30) DEFAULT 'Activo',
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT contratos_estado_check CHECK (estado IN ('Activo', 'Finalizado', 'Cancelado')),
    CONSTRAINT contratos_tipo_contrato_check CHECK (tipo_contrato IN ('Venta', 'Alquiler', 'Anticretico'))
);

CREATE TABLE IF NOT EXISTS public.password_reset_tokens (
    email character varying(255) NOT NULL PRIMARY KEY,
    token character varying(255) NOT NULL,
    created_at timestamp(0) without time zone
);

CREATE TABLE IF NOT EXISTS public.prospectos (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    agente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    nombre character varying(255) NOT NULL,
    telefono character varying(20),
    email character varying(255),
    propiedad_id bigint REFERENCES public.propiedades(id) ON DELETE SET NULL,
    etapa character varying(50) DEFAULT 'Nuevo',
    notas text,
    fecha_contacto date DEFAULT CURRENT_DATE,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT prospectos_etapa_check CHECK (etapa IN ('Nuevo', 'Contactado', 'Interesado', 'Negociando', 'Cerrado', 'Perdido'))
);

CREATE TABLE IF NOT EXISTS public.registro_actividad (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id bigint REFERENCES public.usuarios(id) ON DELETE SET NULL,
    nombre character varying(255),
    correo character varying(255),
    rol character varying(255),
    accion character varying(255) NOT NULL,
    descripcion text,
    ip character varying(45),
    fecha_hora timestamp without time zone DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.resenas (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    cliente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    puntuacion integer NOT NULL,
    comentario text,
    fecha timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT resenas_puntuacion_check CHECK (puntuacion >= 1 AND puntuacion <= 5)
);

CREATE TABLE IF NOT EXISTS public.seguimientos (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    agente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    cliente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    tipo_contacto character varying(50) NOT NULL,
    descripcion text NOT NULL,
    fecha date NOT NULL DEFAULT CURRENT_DATE,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT seguimientos_tipo_contacto_check CHECK (tipo_contacto IN ('Llamada', 'Correo', 'Visita', 'WhatsApp', 'Otro'))
);

CREATE TABLE IF NOT EXISTS public.sessions (
    id character varying(255) NOT NULL PRIMARY KEY,
    user_id bigint,
    ip_address character varying(45),
    user_agent text,
    payload text NOT NULL,
    last_activity integer NOT NULL
);

CREATE TABLE IF NOT EXISTS public.solicitudes_visita (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    cliente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    fecha_solicitada timestamp(0) without time zone NOT NULL,
    mensaje text,
    estado character varying(255) NOT NULL DEFAULT 'Pendiente',
    created_at timestamp(0) without time zone,
    updated_at timestamp(0) without time zone,
    deleted_at timestamp(0) without time zone,
    CONSTRAINT solicitudes_visita_estado_check CHECK (estado IN ('Pendiente', 'Aceptada', 'Rechazada', 'Completada'))
);

CREATE TABLE IF NOT EXISTS public.ubicaciones (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    ciudad character varying(100) DEFAULT 'Santa Cruz',
    zona character varying(150),
    direccion text,
    latitud numeric(10,7),
    longitud numeric(10,7),
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS public.users (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name character varying(255) NOT NULL,
    email character varying(255) NOT NULL UNIQUE,
    email_verified_at timestamp(0) without time zone,
    password character varying(255) NOT NULL,
    remember_token character varying(100),
    created_at timestamp(0) without time zone,
    updated_at timestamp(0) without time zone
);

-- ÍNDICES ADICIONALES
CREATE INDEX IF NOT EXISTS sessions_last_activity_index ON public.sessions (last_activity);
CREATE INDEX IF NOT EXISTS sessions_user_id_index ON public.sessions (user_id);

-- =====================================================
-- CICLO #5: NUEVAS TABLAS PARA INTELIGENCIA Y ASISTENCIA
-- =====================================================

-- 1. NOTIFICACIONES (CU23)
CREATE TABLE IF NOT EXISTS public.notificaciones (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    usuario_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    propiedad_id bigint REFERENCES public.propiedades(id) ON DELETE SET NULL,
    tipo character varying(50) NOT NULL,
    mensaje text NOT NULL,
    leida boolean DEFAULT false,
    fecha_envio timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT notificaciones_tipo_check CHECK (tipo IN ('cambio_estado', 'nueva_propiedad', 'recordatorio', 'recomendacion'))
);

CREATE INDEX IF NOT EXISTS idx_notificaciones_usuario_id ON public.notificaciones (usuario_id);
CREATE INDEX IF NOT EXISTS idx_notificaciones_leida ON public.notificaciones (leida);
CREATE INDEX IF NOT EXISTS idx_notificaciones_fecha_envio ON public.notificaciones (fecha_envio);

-- 2. RECOMENDACIONES (CU24)
CREATE TABLE IF NOT EXISTS public.recomendaciones (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cliente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    puntuacion_recomendacion numeric(5,2) DEFAULT 0,
    vista boolean DEFAULT false,
    fecha_recomendacion timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT recomendaciones_unica UNIQUE (cliente_id, propiedad_id)
);

CREATE INDEX IF NOT EXISTS idx_recomendaciones_cliente_id ON public.recomendaciones (cliente_id);
CREATE INDEX IF NOT EXISTS idx_recomendaciones_fecha ON public.recomendaciones (fecha_recomendacion);

-- 3. PREDICCIONES (CU25)
CREATE TABLE IF NOT EXISTS public.predicciones (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    zona character varying(255),
    tipo_propiedad character varying(255),
    probabilidad_venta numeric(5,2) DEFAULT 0,
    dias_estimados_venta integer,
    tendencia character varying(50),
    fecha_prediccion timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT predicciones_tendencia_check CHECK (tendencia IN ('Alta', 'Media', 'Baja', 'Estable'))
);

CREATE INDEX IF NOT EXISTS idx_predicciones_zona ON public.predicciones (zona);
CREATE INDEX IF NOT EXISTS idx_predicciones_tendencia ON public.predicciones (tendencia);
CREATE INDEX IF NOT EXISTS idx_predicciones_fecha ON public.predicciones (fecha_prediccion);

-- 4. HISTORIAL_CLIENTE (CU24 y CU25)
CREATE TABLE IF NOT EXISTS public.historial_cliente (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cliente_id bigint NOT NULL REFERENCES public.usuarios(id) ON DELETE CASCADE,
    propiedad_id bigint NOT NULL REFERENCES public.propiedades(id) ON DELETE CASCADE,
    accion character varying(50) NOT NULL,
    fecha_accion timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT historial_cliente_accion_check CHECK (accion IN ('vista', 'visita_solicitada', 'visita_realizada', 'favorito', 'recomendacion_vista'))
);

CREATE INDEX IF NOT EXISTS idx_historial_cliente_id ON public.historial_cliente (cliente_id);
CREATE INDEX IF NOT EXISTS idx_historial_fecha ON public.historial_cliente (fecha_accion);
